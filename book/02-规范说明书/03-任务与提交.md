## 5. 任务

### 5.1 定义

```ts
type TaskOutcome<R> =
  | { readonly status: "completed"; readonly result: R }
  | { readonly status: "failed"; readonly error: TaskOutcomeError; readonly result?: R }
  | { readonly status: "aborted"; readonly reason?: string; readonly result?: R }
  | { readonly status: "orphaned"; readonly reason: string }
  | { readonly status: "faulted"; readonly error: TaskOutcomeError };

type JoinPolicy = "failFast" | "allSettled";

type TaskState<S, R> =
  | { readonly status: "pending"; readonly checkpoint: S }
  | { readonly status: "running"; readonly checkpoint: S }
  /** Parked without an invocation until every task in `on` is terminal; then resumes at `checkpoint` (section 5.5). */
  | {
      readonly status: "waiting";
      readonly checkpoint: S;
      readonly on: readonly TaskId[];
      readonly policy: JoinPolicy;
    }
  /** Outcome decided; terminal once no ordinary owned work below is live (section 5.5). Runs no more code. */
  | { readonly status: "completing"; readonly outcome: TaskOutcome<R> }
  | { readonly status: "terminal"; readonly outcome: TaskOutcome<R> };

/** Who owns a task: its conversation (a top-level task) or another task of the same conversation. */
type TaskOwnership = { readonly kind: "conversation" } | { readonly kind: "task"; readonly taskId: TaskId };

type TaskRecord<I, S, R> = {
  readonly id: TaskId<R>;
  readonly conversationId: ConversationId;
  readonly kind: string;
  readonly version: number;
  readonly input: I;
  /** Owning task; absent for a task its conversation owns. Immutable. */
  readonly owner?: TaskId;
  readonly background: boolean;
  readonly abortRequested: boolean;
  /** Wall clock at the first change to `running`; kept through waits and recovery. */
  readonly startedAt?: number;
  /** Wall clock at the change to `terminal`. */
  readonly endedAt?: number;
} & (
  | {
      readonly state: Extract<TaskState<S, R>, { status: "pending" | "running" | "waiting" }>;
      readonly memos?: Readonly<Record<string, JsonValue>>;
    }
  | {
      readonly state: Extract<TaskState<S, R>, { status: "completing" | "terminal" }>;
      readonly memos?: never;
    }
);

type RunningTask<I, S, R> = TaskRecord<I, S, R> & {
  readonly state: Extract<TaskState<S, R>, { status: "running" }>;
};

/**
 * State a task commits for itself: a replacement checkpoint, a wait, or its outcome. A returned `terminal` state
 * becomes `completing` while ordinary owned work below is live (section 5.5).
 */
type NextTaskState<S, R> = Extract<TaskState<S, R>, { status: "running" | "waiting" | "terminal" }>;

interface HookRunner<H extends object> {
  each<K extends keyof H>(name: K, invoke: (handler: NonNullable<H[K]>) => void | Promise<void>): Promise<void>;
}

type PhaseHandler<I, P, S, R, H extends object> = (
  task: RunningTask<I, P, R>,
  runtime: TaskRuntime<I, S, R, H>,
  context: Context,
) => Promise<void>;

interface TaskRuntime<I, S, R, H extends object> extends DocumentObserver, DocumentReader {
  readonly taskId: TaskId<R>;
  readonly conversationId: ConversationId;
  readonly signal: AbortSignal;
  /** Registry snapshot of the current phase; refreshed at every phase boundary. */
  readonly registry: RegistrySnapshot;
  /** The task's conversation's agent, resolved once for this phase (section 7.1). */
  agent(context: Context): Promise<Agent>;
  /** `HarnessOptions.settings`, resolved at each access. */
  readonly settings: Settings;
  readonly models: Models;
  readonly hooks: HookRunner<H>;
  /** Calls `HarnessOptions.env` for the task's conversation; rejects with its error. */
  env(context: Context): Promise<ExecutionEnv | undefined>;

  commit(
    change: (
      tx: Tx,
      current: RunningTask<I, S, R>,
    ) => NextTaskState<S, R> | undefined | Promise<NextTaskState<S, R> | undefined>,
    context: Context,
  ): Promise<void>;

  memo<T extends JsonValue>(name: string, context: Context): Promise<T | undefined>;
  memo<T extends JsonValue>(name: string, candidate: T, context: Context): Promise<T>;
  conversation(id: ConversationId, context: Context): Promise<ConversationHandle | undefined>;
  /** Committed task record. */
  getTask<T>(id: TaskId<T>, context: Context): Promise<TaskRecord<JsonValue, JsonValue, T> | undefined>;
  /** Terminal receipt; rejects when the invocation ends. */
  waitForTask<T>(id: TaskId<T>, context: Context): Promise<SettledTask<T>>;
  /** Outcomes of terminal tasks, in order; rejects when one is not terminal. Used after a wait (section 5.5). */
  outcomes<T>(ids: readonly TaskId<T>[], context: Context): Promise<TaskOutcome<T>[]>;
  /** Committed entry visible from the task's conversation. */
  entry(id: EntryId, context: Context): Promise<EntryRecord | undefined>;
  entry<D extends JsonValue>(token: Entry<D>, id: EntryId, context: Context): Promise<TypedEntry<D> | undefined>;
  /** Committed raw active transcript and model context, optionally cut off at `at`. */
  context(conversationId: ConversationId, context: Context, options?: { readonly at?: EntryId }): Promise<ContextView>;
  /** The Harness clock. */
  now(): number;
  /** Forward a non-fatal failure to `HarnessOptions.onReport`. */
  report(error: unknown): void;
  sleep(until: number, context: Context): Promise<void>;
}

type TaskDefinition<I, S extends { phase: string }, R, H extends object> = {
  readonly name: string;
  readonly version: number;
  initial(input: I): S;
  readonly phases: {
    [P in S["phase"]]: PhaseHandler<I, Extract<S, { phase: P }>, S, R, H>;
  };
  abort(task: RunningTask<I, S, R>, runtime: TaskRuntime<I, S, R, H>, context: Context): Promise<void>;
  migrate?(input: JsonValue, checkpoint: JsonValue, fromVersion: number): {
    input: I;
    checkpoint: S;
  };
  readonly hooks?: H;
};

interface Task<I, S extends { phase: string }, R, H extends object> {
  readonly definition: TaskDefinition<I, S, R, H>;
}

type TaskOptions = {
  /** Required: a task always names its owner (section 5.5). */
  readonly ownership: TaskOwnership;
  /** Default: the owner task's conversation, or the transaction's bound conversation. */
  readonly conversationId?: ConversationId;
  /** Conversation-owned tasks only: excluded from conversation abort and idle, and from cascades. */
  readonly background?: boolean;
};

function defineTask<I, S extends { phase: string }, R, H extends object = {}>(
  definition: TaskDefinition<I, S, R, H>,
): Task<I, S, R, H>;
```

`TaskId<R>` 就是数字形式的任务（task）ID 本身，附带一个已被擦除的结果类型标记（brand）。它取代了单独的任务引用包装器：`createTask()` 返回 `TaskId<R>`，带类型的等待从该 ID 推断 `R`，而与结果无关的操作则接受 `TaskId<unknown>`。任务的 `conversationId` 在创建之后不可变；若一次替换想把已有任务移动到另一个会话，会在 Storage 受理（admission）之前就被拒绝，这样持久化的「会话—拥有者」边就不会失效。

阶段映射（phase map）是穷尽的，并且按阶段收窄类型。一个处理器可以在一个外部效果（effect）前后执行多次提交（commit），但每个持久化的检查点（checkpoint）都是一次完整替换。`TaskRuntime.commit()` 在调用回调之前，会先在 Session 线上重读当前的持久化任务并做门控。当调用（invocation）已经结束、Harness 正在关闭、任务已处于终态（terminal），或 run 调用的任务带有中止标记（abort mark）时，它都会拒绝。它的 `tx.createTask()` 默认使用任务所在的会话，且它追加的每个条目（entry）都会把该任务记为 `byTaskId`。当回调返回一个状态时，运行时会在同一次提交中替换任务的状态，因此检查点或结果（outcome）与回调写入的条目、文档及子任务是原子的，并会针对任务的检查点类型与结果类型做类型检查。不返回任何东西则状态保持不变。`terminal` 或 `completing` 状态会丢弃备忘（memo）。`waiting` 状态即使检查点未变也会结束本次调用；当 `on` 指向一个不存在的任务、任务自身，或其拥有者链（owner chain）上的任务——后者永远不可能先完成——时，它会拒绝；空的 `on` 则在下一轮调度时恢复。`pending` 永远不会被返回；只有对账（reconciliation）与移交（handover）才会写入它。

`Tx` 没有任务替换操作。任务只能通过自己的运行时（runtime）改变自身的状态。调度器（scheduler）掌管预留（reservation）、对账、移交、故障（fault）与孤儿化（orphaning）；`abortTask()` 掌管中止标记。其他代码用 `abortTask()` 停止任务，用 `waitForTask()` 读取其结果。

```ts
// Intent, effect, outcome.
prepare: async (task, runtime, context) => {
  await runtime.commit(() => ({ status: "running", checkpoint: { phase: "charge", key: newKey() } }), context);
},
charge: async (task, runtime, context) => {
  const receipt = await payments.charge(task.state.checkpoint.key); // idempotent by key
  await runtime.commit(async (tx, current) => {
    const entry = await tx.appendEntry(current.conversationId, receiptEntry(receipt));
    return { status: "terminal", outcome: { status: "completed", result: { entryId: entry.id } } };
  }, context);
},
// The abort handler decides the outcome; returning without one faults the task.
abort: async (task, runtime, context) => {
  await payments.cancel(task.state.checkpoint);
  await runtime.commit(() => ({ status: "terminal", outcome: { status: "aborted", reason: "user" } }), context);
},
```

`memo(name, candidate)` 是一次带门控的提交；`memo(name)` 读取已提交的记录。`sleep(until)` 与 Harness 的 `now` 时钟比较，并在调用收到信号或其上下文（context）被取消时拒绝。通过运行时获取的 watch 会在调用结束时停止。`snapshot()`/`snapshotAsOf()` 读取已提交的文档，例如用于提供 `PromptInput.read`（第 7.4 节）。`getTask()` 与 `entry()` 各用一次查找读取已提交的记录；`waitForTask()` 等待一个终态回执（terminal receipt），例如工具创建的子任务。`context()` 在 Session 线上捕获其边界，并像 `Conversation.context()` 一样从线外的不可变条目推导视图。调度器会把每个对话最近一次读取的范围与视图保存在内存中：只要头部标记相同，其任何任务稍后的读取就只扫描并推导该范围尾部之后新提交的条目。对话在忙碌期间一直保留它们，空闲后保留 `contextRetentionMs` 这么久。过期检查发生在任务变化时，也由一个未引用（unreferenced）的定时器执行——该定时器绝不会让进程保活；在无法让定时器未引用的环境（如 Cloudflare Workers）中，只靠任务变化检查过期。被保留的条目是冻结的，每次读取返回各自的视图数组。与所有运行时操作一样，这些操作在调用结束之后都会拒绝。

预留会把 `pending`——以及在可以恢复之后的 `waiting`——持久化地改为 `running`。一次调用按顺序运行各阶段处理器（phase handler）；检查点提交保持 `running` 不变。在每个阶段之前（包括第一个阶段），调度器都会在 Session 线上运行一次 step 回调。它读取已提交的任务，并同步地对刚返回的阶段应用下面第一条匹配的规则。任何故障或移交都会写入同一次提交，连同针对调度器所写结果的 Harness 清理（第 5.4 节），后者可以在该回调内部 await 文档访问。当某条规则要求停止时，调用就在那里结束。因此，调用先前排入队列的运行时提交，要么在该 step 之前落地并被计入，要么在它之后到达线上而被拒绝。在第一个阶段之前，只有规则 1–3 适用：

1. `terminal`、`completing` 或 `waiting`：停止。
2. Session 正在关闭：停止；保留检查点与任何中止标记，供重开时使用。
3. run 模式下带有持久化中止标记：结束并加入（join）run 调用；一旦任务的普通自有工作（ordinary owned work）不再存活（第 5.5 节），就启动一个全新的中止调用（abort invocation）。
4. 未捕获的错误：写入终态 `faulted`。
5. 检查点发生变化（包括同一阶段内的进展）：刷新注册表快照（registry snapshot），或者移交（hand over，第 5.4 节），或者在同一任务调用中调用阶段处理器。
6. 检查点未变：写入终态 `faulted`，因为没有取得任何持久化的进展。

中止调用只运行一次其中止处理器（abort handler）。在其结束（settle）之后，一次 step 会应用规则 1、2 和 4；处理器若在返回时没有给出终态结果，任务即被判定为故障。当 Storage 拒绝某次 step 的故障或移交写入时，调用仍然结束；任务保持 `running`，下一次预留会再次运行它。

打开（open）时，对运行中任务的对账会把幸存的 `running` 任务改回 `pending`，保留其检查点与中止标记；`waiting` 任务保持等待，`completing` 与 `waiting` 任务会被重新评估（第 5.5 节）。任务迁移（migration）在预留时运行，与 `pending` 或 `waiting` -> `running` 的转换原子地完成。一个回调处理所有受支持的旧版本。定义缺失、比存储版本更旧、或迁移失败的任务会保持 `pending` 或 `waiting` 并处于阻塞（blocked）状态，直到安装了合适的定义，或该任务被中止（第 5.4 节）。`close()` 标记运行时正在关闭，封闭（seal）受理与预留，向各调用发信号，并终止各状态与 watch。在 Session 线之外，它先让已受理的提交结算、加入各调用，然后再关闭存储。已在运行的 watch 回调仍归调用者所有。此后的运行时提交都会拒绝，且 close 不写入任何任务结果。关闭不会启动新的阶段调用或中止调用。它不设置中止标记，不把任务终态化（terminalize），不退役（retire）任务文档，也不发布文档退役。宿主层会在关闭 Harness 之前撤回服务并分离客户端，例如释放其 Chord facet host（参见 Chord 使用指南）；重新连接到重开的 Session 会水合（hydrate）最后一次提交的状态，并从其持久化检查点恢复。

### 5.2 效果三明治

```text
commit intent phase
perform external effect
commit outcome or next phase
```

在意图阶段重新打开，意味着外部效果可能已经发生。阶段处理器会安全地重试、轮询一个外部句柄（handle），或记录中断。deferred 提供方（deferred provider）由一个持久化阶段表示，其中包含其句柄与下次轮询时间。

运行时持有的备忘（memo）是存储在活跃任务信封（envelope）中的小体积、先写者胜（first-writer-wins）的值。候选值的插入与读取胜出者在同一次 Session 提交中完成，因此并发的候选者会得到同一个持久化的胜者。备忘在检查点之间存续，并在终态替换中消失。大批量的进度数据应放进文档。

### 5.3 终态任务

终态任务记录就是持久化的结果回执（receipt）。其结果可以直接包含一个小值，或引用一个条目：

```ts
{ status: "completed", result: { entryId: toolResultId } }
```

一次终态转换会原子地：

1. 写入终态任务记录。
2. 追加所有结果条目。
3. 退役所有归属于该任务范围的文档。
4. 兑现（resolve）所有由该任务结算的提交（submission）。

若任务的普通自有工作仍在存活，这一转换会在 `completing` 持有点（hold）处拆分（第 5.5 节，规则 4）：任务自己写下的结果所伴随的写入在持有点落地，而第 1、3 项及其等待者（waiter）则在最终提交中跟进。

执行检查点与备忘会从终态表示中消失。终态记录仍然可查询，供等待、等待者、检查（inspection）与重开使用。

Session 会用它打开时的时钟（Harness 用 `HarnessOptions.now`，默认 `Date.now`）在任务记录上标记生命周期时间：第一次变为 `running` 时标 `startedAt`，变为 `terminal` 时标 `endedAt`。一旦设置，两者都会从被替换的记录延续，因此 `startedAt` 能挺过等待、`completing` 暂停与重开，两者之间的时间跨度也把这些包含在内。从未运行过的任务——例如在首次预留之前就被孤儿的任务——只有 `endedAt`；更早版本写入的记录两者都没有，而升级时仍在运行的任务会在下一次运行时获得 `startedAt`。`inspect()` 会在每个活跃任务的记录上显示它们。它们是生命周期时间，不是执行时间：工具的执行时间是其结果的 `durationMs`（第 7.3 节）。

任务之间不存在独立（free-standing）的依赖关系：顺序来自一个任务对其他任务的等待（第 5.5 节）。中止标记让排队中的工作得以进入其中止处理器；若其定义不可用，则被结算为 `orphaned`。

### 5.4 调度器、中止与归属

调度器串行地预留符合条件的任务，然后在 Session 线之外运行处理器。一个内存中的 `TaskInvocation` 包含模式（mode）、中止控制器（abort controller）与完成 promise。

中止协议（abort protocol）：

```text
commit abortRequested
signal and join active run invocation
wait until ordinary owned work is no longer live (section 5.5)
start a fresh abort invocation
abort handler commits terminal outcome
```

在持久化中止标记出现之后，run 调用不得再提交。每个运行时操作在其所属调用结束之后都会拒绝——即使 Session 仍然打开；并且无论调用因何种原因结束，其信号与处理器上下文都会中止。调用只在其最后一个处理器返回之后才结束，因此这种取消只影响游离的残留操作，例如未被 await 的 `waitForTask()`，或以处理器上下文发起的 fetch——它们已经无法再写入任何东西。从一个阶段处理器返回并不会结束一个还要继续进入下一阶段的调用。调用的模式是易失的，在重开时从持久化标记推导。取消某个调用者的 `Context` 只会取消那一次调用或等待；除非被调用的 API 提交了一个中止标记，否则它不会持久化地中止共享的工作。

任务可以创建自己拥有的会话与任务（第 5.5 节）。会话是持久化的作用域（scope）；任务是活跃工作的计量单位，由空闲（idle）计数、由中止标记。历史父条目与归属遍历（ownership traversal）无关。

普通遍历（ordinary traversal）从一个被显式指定的会话出发，访问其会话拥有的任务，跟随每个非后台（non-background）任务所拥有的任务，并跟随每个非后台任务所拥有的会话。即使拥有者（owner）已经终态，它仍会跟随拥有者边（owner edge）；但后台（background）任务是一个边界：普通遍历会跳过该任务及其完整的所属子树。直接的会话操作从该会话内部出发，无论其拥有者是谁。直接中止一个存活的后台任务会包含该任务本身，并跟随其普通自有子树；嵌套的后台拥有者仍是边界。`close()` 会停止每一个调用——前台与后台——而不写入任何结果，因此工作在重开时恢复。必须持久化地取消一切的宿主，可以对 `inspect()` 列出的每个存活任务执行中止。

`Conversation.abort()` 会撤回排队的输入，并标记由普通遍历选中的、存活的非后台任务。传入 `{ background: true }` 时它会跨越后台边界：在该操作被受理时，它会标记无视后台标志的遍历所能到达的每一个存活任务，并撤回它到达的每个会话中排队的输入。随后它等待它标记的每个任务都达到终态、且该会话普通地空闲；此后创建的后台工作既不会被标记，也不会被等待。`Conversation.waitForIdle()` 等待该遍历中不再包含存活的非后台任务。Harness 空闲从每个无主（ownerless）会话根执行同样的遍历，而不是全局地统计任务，因此后台拥有者之下的普通工作不会阻塞它。显式的 `waitForTask()` 等待其引用的任务，无论该任务是否为后台。

中止标记以幂等方式级联（cascade）到前台拥有的工作。持久化的取消意图（cancellation intent）是存活拥有者自己的记录：它的中止标记，或一个持有的 `completing` 结果且该结果不是 `completed`（第 5.5 节）。终态的拥有者绝不级联：它只在其普通自有工作排空（drain）之后才成为终态，因此其意图已经施加完毕；之后在其拥有的会话中启动的工作——例如用户盘问一个已完成的 subagent——会正常运行，并且可以被从上往下的普通遍历到达。其拥有的会话中排队的输入保持排队，就像一次失败的 run 之后那样（第 6 节）。调度器在稍后的一次对账提交中推导这些标记，并在打开时再次推导，因此期间的崩溃不会丢失任何东西；已经排在 Session 线上的工作可以先完成一次提交。每一个存活的非后台任务，只要沿其拥有者链向上穿过拥有任务与拥有会话、在不先经过一个自身没有取消意图的后台任务的前提下，到达一个已被取消的存活拥有者，就会得到一个中止标记——级联之后在那里创建的工作也不例外。在独立的提交中推导标记，可以把它们排除在拥有者的提交之外——后者可能已经写入了所涉及的任务。级联到达的每个会话都按 `Conversation.abort()` 处理：其排队的输入提交变为 `unanswered`（原因 `aborted`）并离开其收件箱（inbox），而排队中的写入保持不变。级联从被取消任务的下方开始，因此该任务自己的会话保留其队列。持有的结果 `failed`、`faulted`、`orphaned` 与 `aborted` 记录与中止标记相同的持久化取消意图；`completed` 则不然。会话记录与拥有者边从不随任务一起退役。活跃的调用在提交之后收到信号。终态回执保证该任务的普通自有工作已排空；其下的后台工作可以继续。

每次 run 调用启动时，都会按 `TaskRecord.kind` 从调度器当前的注册表快照中解析任务的定义（definition）：

```text
definition missing                 -> stay pending, blocked: missing_task
stored version newer than it       -> stay pending, blocked: task_too_old
stored version older than it       -> migrate(input, checkpoint, storedVersion)
  failure                           -> stay pending, blocked: migration_failed
  success                           -> commit migrated record + running atomically
equal version                       -> commit running
```

中止调用以同样的方式解析定义。当定义能够接手该任务时，其中止处理器会运行；否则该任务被孤儿化，如下所述。

失败的迁移会通过 `onReport` 报告一次，只有当注册表为该任务的种类（kind）解析出另一个不同的定义对象时才会重试。

Harness 绝不会仅仅因为注册表代码缺失或不兼容就把任务终态化——无论在打开时还是之后。没有中止标记的阻塞任务保持其持久化记录不变，仍然存活，仍会阻塞普通的空闲等待，并且在注册表每次变化时都会被重新考虑。阻塞原因是推导出来的运行时状态，不是持久化的任务状态。文档迁移仍然由访问驱动。

中止一个阻塞的任务无法运行其中止处理器，因为那段代码缺失，或者无法接手该任务。因此 Harness 把它结算为终态 `orphaned` 而不是 `aborted`：`aborted` 意味着任务自己的中止处理器已经运行并决定了结果，而 `orphaned` 意味着没有任何任务代码运行过，因此任务启动的外部效果可能没有被清理。孤儿化的 `reason` 就是阻塞原因（`missing_task`、`task_too_old` 或 `migration_failed`）。当 `abortTask()` 发现没有活跃调用、没有存活的普通自有工作、且当前快照无法接手该任务时，标记提交会直接把它结算掉；否则调度器会在本应预留中止调用的时候结算它，而那发生在自有工作排空之后（第 5.5 节），因此孤儿化结果永远不会处于持有状态。只有中止（直接中止、经会话中止或经级联中止）才会孤儿化一个任务；仅凭定义缺失绝不会。孤儿化提交会执行任务代码无法完成的清理：受影响的输入提交变为 `unanswered` 并携带原因，任何匹配的活跃 run 控制被清除，任务范围的文档退役。终态任务记录与 `unanswered` 提交携带该原因；唯一写入 transcript 的条目是把一个已提交的 generation partial 转换掉（见下文）。使一个 run 任务故障会以 `faulted` 结果执行同样的控制/提交清理；若故障发生时普通自有工作仍存活，则先持有，其清理在最终提交中执行（第 5.5 节，规则 4）。

调度器对 run 与任务种类一无所知。拥有提交、run 控制与内置任务的 Harness 向它提供了一个钩子（hook），调度器在自己写入的结果（`faulted` 与 `orphaned`）被终态化的那次提交中调用它——也就是持有点之后的最终提交。该钩子忽略种类不是内置 run、tool 或 compaction 的任务，因此它绝不会在其他地方创建 `pi.live`。当 `pi.live.run` 指向该任务时（第 8 节），它会结算该 run：已提交的 generation partial 变成一条 `aborted` 的 `pi.assistant` 条目，与 generation 中止处理器的转换完全一致，因此 transcript 保留模型已产出的内容，且 `pi.usage` 统计其开销（调度器的提交没有任务范围，因此该条目没有 `byTaskId`）；其输入变为 `unanswered`，原因为 `faulted`（detail：错误消息）或阻塞原因；`run`、`generation` 与 `tools` 被移除。故障来自任务的 bug 或畸形的提供方数据，例如响应中的非 JSON 值，或被 Storage 无效果地拒绝的提交（`StorageRejected`）。不确定的存储失败会让 Session 中毒（poison），且不写入任何结果。对 `pi.tool` 任务，它把该任务的工具槽位（tool slot）标记为 `done` 且不带条目；run 继续进行，上下文推导会合成（synthesize）缺失的结果（第 2.1 节）。对 `pi.compaction` 任务，它移除该任务的压缩状态（第 8.7 节）。任务为自己提交的结果自行完成结算。Harness 还提供 `runtime.hooks` 与 `runtime.agent()` 背后按阶段的 agent 解析（第 7.1 节）；调度器只是把每个阶段的注册表快照传给它。

在每个正常的阶段边界（第 5.1 节，规则 5），step 会刷新调用的注册表快照。如果按名字解析出的任务定义与调用启动时的定义不是同一个对象，且新定义能够预留该任务（相同版本，或带有 `migrate` 的更高版本），调用就移交：step 把任务连同其检查点、备忘与中止标记提交回 `pending` 并结束调用，下一次预留则在新定义下启动一个全新的调用，并应用上述预留规则。当定义缺失或无法预留该任务时，调用继续在旧定义下运行，每个解析到的定义通过 `onReport` 报告一次，并在其下一个边界重新考虑。一个从不结束（settle）的处理器永远不会移交。

### 5.5 结构化并发

任务与会话构成一棵归属树（ownership tree）。每个任务在创建时都要指明其拥有者，即 `tx.createTask(task, input, { ownership })` 携带必需的 `TaskOwnership`：其会话（`{ kind: "conversation" }`，即顶层任务）或某个任务（`{ kind: "task", taskId }`，即子任务）。每个会话要么无主，要么由一个任务拥有（第 2.2 节）。子任务总是位于其拥有者的会话中；只有被拥有的会话才会跨越会话边界。拥有者边不可变。

任务 `T` 的**普通自有工作（ordinary owned work）**是普通遍历在 `T` 之下能到达的每个任务：`T` 拥有的任务、`T` 拥有的会话中的任务，以及（传递地）其中每个非后台任务的同样集合。后台任务及其下的一切都被排除。会话拥有的任务可以是后台任务；子任务则不可以。

从这棵树可以推出两条规则。

**中止向下流动（abort flows down）**（第 5.4 节）。存活拥有者的取消意图——它的中止标记，或一个持有的非 `completed` 结果——会标记其普通自有工作。后台任务是边界，除非被直接中止，或被 `Conversation.abort(context, { background: true })` 中止。

**自有工作不会比拥有者的完成更长寿。** 一个任务只有在其普通自有工作排空之后才算完成：

1. 任务提交的终态，或调度器写入的终态结果（`faulted`、`orphaned`），若在任务的普通自有工作仍然存活时出现，则改为以 `{ status: "completing", outcome }` 存储。调度器会在稍后的提交中写入最终的 `terminal` 记录——一旦不再有任何普通自有工作存活；这一判定在每次提交之后都会重新进行，包括持有点之后创建的工作。工作是否存活按该提交的候选记录（candidate records）判定，因此完成提交自身创建的工作——例如该任务拥有的会话中的一个任务——会让它继续持有。打开时会重新评估每个 `completing` 任务。
2. 持有的结果是最终的：不再运行任何阶段、运行时提交或中止处理器，不需要定义，任务永远不会被预留或迁移，对它调用 `abortTask()` 只会标记它并返回 `marked`，这会取消下方的工作。最终记录保留持有的结果与标记。
3. 持有的非 `completed` 结果就是取消意图，因此下方的工作被中止、排空，然后任务变为终态。持有的 `completed` 结果则等待这些工作正常完成。
4. 写入在持有点处拆分。任务写入的终态提交会在持有点落地其所有其他写入，例如工具的结果条目与槽位。只有记录的终态、任务文档的退役与任务等待者被推迟到最终提交。调度器写入的结果在持有点只写记录；其 Harness 清理（第 5.4 节）在最终提交中运行，因此故障的 run 任务在其工具排空之前一直保留 `pi.live.run`。
5. 等待者、空闲等待、检查与普通遍历都把 `completing` 任务视为存活，直到其最终提交。

新的自有工作需要一个存活的拥有者：带任务归属的 `createTask` 与带任务归属的 `createConversation` 会拒绝在提交的最终候选中处于 `completing`、`terminal` 或带有中止标记的拥有者（第 3.3 节）。会话在其拥有者完成之后仍然可用：在那里启动的新 run——例如为了盘问一个已完成的 subagent——是该会话的普通工作，可以被从上往下的遍历到达，而不属于任何一次完成的一部分。

**等待（waiting）。** 一个阶段可以提交 `{ status: "waiting", checkpoint, on, policy }`。任务停止、不再有调用，并在 `on` 中的每个任务都达到终态后从 `checkpoint` 恢复。`on` 可以指向任何任务，包括已经终态的任务；不属于等待任务所有的任务——即其 `owner` 不是该等待任务——要求 `policy: "allSettled"`。`on` 不得指向不存在的任务、任务自身，或其拥有者链上的任务（第 5.1 节）。使用 `failFast` 时，`on` 中第一个持有或以非 `completed` 结果结束的任务，会在调度器的下一次对账提交中给 `on` 里其他每个存活任务一个中止标记。等待任务自身不会被标记：一旦 `on` 全部终态，它就恢复，并用 `runtime.outcomes()` 读取它们的结果。任务可以在等待之前创建子任务并继续运行阶段，也可以按顺序对其子任务的任意子集依次等待。

**中止顺序自底向上。** 一个任务的中止调用只在其普通自有工作不再存活之后才启动，因此中止处理器看到的下方结果都是最终结果。等待任务上的中止标记让它可以依据这条同样的规则提前离开等待、进入其中止处理器；其 `on` 中不属于它的任务不会被等待。顺序按已提交的记录判定：子任务在其终态提交之后运行的代码不在顺序约束之内。中止处理器不得返回 `waiting`，也不能创建被拥有的子任务，因为其任务已带有中止标记（第 3.3 节）：它可以内联补偿，或创建后台的会话拥有任务——任何级联都到不了那里——并用 `runtime.waitForTask()` 等待它们，这会占住（hold）其调用。

以示例任务 `Checkout`（以 `failFast` 等待四个 `Payment` 子任务）为例的级联：一张过期的卡使其支付失败；三个存活的支付任务得到中止标记，并在各自的中止处理器中退款；当四个全部终态后，`Checkout` 恢复，通过 `outcomes()` 读到 `[aborted, failed, aborted, aborted]`，然后决定自己的结果。而 `abortTask(Checkout)` 则是标记 `Checkout`，级联标记存活的支付任务，它们的中止处理器先运行，然后 `Checkout.abort` 运行并看到它们的最终结果。

## 6. 提交与收件箱

提交（submission）记录支撑着可等待的宿主对象。受理是 Harness 内部的；边界用 `tx.placeSubmission()` 放置（place）排队的提交，run 任务用 `tx.settleSubmission()` 结算（settle）它们所回答的输入。排队中的提交在内置的收件箱（inbox）文档中等待——这是一个由带标签条目构成的有序列表：

```ts
type InboxItem =
  | { readonly id: SubmissionId; readonly mode: "steer" | "followUp"; readonly content: UserInput }
  /** `entry` is the write's `EntryDraft`, stored as plain JSON. */
  | { readonly id: SubmissionId; readonly mode: "write"; readonly entry: JsonObject };

type InboxState = { items: InboxItem[] };
```

| 字段 | 值 |
|---|---|
| kind | `pi.inbox` |
| version | `1` |
| scope/history/fork | 会话，`latest`，`initial` |
| `initial()` | `{ items: [] }` |
| checkpoint | 每当 `items` 为空时写完整基（base） |
| view mount | `docs["pi.inbox"]` |
| created | 随每个 Harness 会话创建（第 2.2 节） |

条目按 ID 顺序排列。排队的输入存储其内容；其 `pi.user` 条目在放置（placement）时取得 Harness 时钟的时间戳。排队的提交属于其会话，因此 fork 从空收件箱开始。

run 控制保存在内置的 live 文档 `pi.live` 中（第 8 节）。其可选的 `run` 值指明当前负责该 run 的任务，以及该 run 已放置的输入提交 ID。`run !== undefined` 定义了 `busy`（忙碌）；空闲文档的获取或创建并不定义忙碌。在 generation 与 tools 相互交接工作的过程中，该值一直存在：`taskId` 指名结算这些输入的 generation，包括其工具轮次（tool round）运行期间，然后是下一个 generation。工具任务从不拥有 run；当前轮次的工具任务列在 `pi.live.tools` 中。这个 ID 列表是可变状态，因为边界会把已放置的 steer（插话）输入加入活跃的 run；每条终态路径都恰好结算列出的这些输入。

受理与终态转换：

| 动作 | 提交状态 | 其他写入 |
|---|---|---|
| 输入，空闲且收件箱为空 | `placed`，带用户条目 | 创建 run 与 generation |
| 写入，空闲且收件箱为空 | `done`，带条目 | 追加条目；无 run |
| 输入或写入，忙碌或收件箱非空 | `queued` | 追加收件箱条目 |
| 边界放置用户条目 | `placed`，带条目 | 把 ID 加入当前或后继 run |
| 边界放置写入 | `done`，带条目 | 追加条目 |
| run 给出回答 | 输入 `done`，带必需的回答条目 | 移除 `run`，或移交给后继 |
| run 失败或其任务中止 | 输入 `unanswered`，带原因 | 移除 `run`；收件箱不变 |
| 撤回排队条目 | `unanswered`，原因 `aborted` | 移除收件箱条目 |
| 过期的头写入 | `unanswered`，原因 `stale` | 移除收件箱条目 |

`requestId` 在一次会话内、在任何写入之前先做去重；把同一个 ID 复用于另一种提交类型会被拒绝。忙碌时的输入若带 `whenBusy: "reject"`，则不写任何记录并报告 `ConversationBusy`。空闲时的输入在 `whenBusy` 为 `steer` 时以 `steer` 模式排队，否则以 follow-up（追问）模式排队。

空闲但收件箱非空的会话——例如一次失败的 run 之后——会把每个新提交排到等待项之后，并在同一次提交中运行一次 final 边界。顺序得以保留：在失败之前排队的 follow-up 会先于新输入被选中。

`Submission` 一直等待到 `done` 或 `unanswered`；abort 只撤回仍在排队的提交，对已放置的输入报告 `already_placed`，对任何已终态的提交报告 `settled`。会话中止会撤回排队的 steer/follow-up 提交，但保留写入（write）以待日后放置。

边界的选择按条目 ID 确定性地进行，使用设置中的 `steeringMode` 与 `followUpMode`（第 2.2 节），并在每个边界于 Session 线上读取：`one-at-a-time` 选取该模式的第一个条目，`all` 选取该模式的全部条目。

| 边界 | 写入 | steer | follow-up |
|---|---|---|---|
| `postTools` | 全部 | 按模式取首个/全部 | 无 |
| `final` | 全部 | 按模式取首个/全部 | 按模式取首个/全部 |

边界先放置其选中的写入（按 ID 顺序），然后放置其选中的用户条目（按 ID 顺序）。因此，在重置（reset）或压缩摘要（compaction summary）之前排队的用户条目会落在它们之后，并在新的上下文中运行。排队的用户条目绝不会因为头（head）而变成过期（stale）。

一个头写入（head write）——无论排队还是放置时立即写入——若其目标早于活跃区间（active range）的起点（即最新的头标记的 `head`），就是过期的：放置它会把那个头所裁掉的历史带回来。同一边界中较早放置的头也算数：在一个排队的重置之后，一个指向更早条目的排队摘要是过期的。`head: "self"` 的写入永不过期。

压缩摘要是头写入（第 8.7 节），因此仅凭这条规则就能按截断位置（cut position）为压缩排序：在活跃区间起点之前截断的摘要是过期的，而在起点或其后截断的摘要则会被放置，无论它何时被选中。示例：后台压缩（background compaction）B 在尾部位置 100 被选中，在 70 处截断。在它做摘要期间，一个阻塞式压缩（blocking compaction）A 在 150 处截断并追加了其摘要。于是 B 的摘要结算为 `stale`，因为 70 早于 150。假如 A 截在 60，B 就会被放置在它之后：B 在被选中时对上下文做的摘要——当时的最新摘要加上 70 之前的条目——覆盖了 A 的摘要覆盖的一切。这只有在头仅由压缩摘要与重置构成时成立，因为它们使更早的截断过期；在压缩做摘要期间放置的应用编辑会丢失，而应用的头写入可以被撤销（第 12 节）。

选中了 `head: "self"` 写入（即重置）的 `postTools` 边界表现得像 `final`：它还会选中 follow-up，且当前 run 以其输入 `unanswered`（原因 `reset`）结束，因为其上下文在得到回答之前就被裁掉了。在 `final` 边界放置的重置跟随在回答之后，因此输入已经是 `done`。

在普通的 `postTools` 边界，即使没有任何排队的触发，generation 也继续进行；被选中的 steer ID 加入该续接（continuation）。终止性或交接性（handoff）的工具轮次（第 8.5 节）则改用 final 行为。在 `final` 边界，当前 run 已放置的输入提交先结算，除非一个 `onYield` 续接让它们保持打开（见下文）；被选中的用户 ID 启动一个后继（successor）generation，以这些 ID 作为新 run 的输入。写入绝不会自行触发 generation。没有用户触发的 final 边界让会话回到空闲——除非有一个 `onYield` 续接（第 8.3 节），它只在边界既未选中用户条目也未选中重置时适用。

只有成功的 run 结束才施加 final 边界：一次回答（answer）、`terminate` 或 `handoff`。失败、run 任务的中止处理器、故障与孤儿会把 run 的输入结算为 `unanswered`，并保持收件箱不动（`Conversation.abort()` 会另行撤回排队的用户条目）；这些排队条目在 `pi.inbox` 中保持可见，直到下一个提交的边界或它们被撤回。

被选中与过期的条目按位置移除，同时被保留条目的顺序保持不变。Chord 的 Astra 操作生成器必须能够在不携带被保留值的情况下表达分散的删除；不得用 ID 替代收件箱的按位置（positional）语义。
